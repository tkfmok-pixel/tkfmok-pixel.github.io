# セクション3：プロジェクトマネジメント 進捗率
``` dataviewjs
// 1. ノートの「生のテキストデータ」をすべて読み込む
// Dataviewの解釈を通さず、書かれている文字そのものを取得します
const content = await dv.io.load(dv.current().file.path);

// 2. 正規表現でチェックボックスを探す
// パターン解説: 
// ^ ... 行の先頭から
// \s* ... スペースがあってもOK
// (?:[-*+]|\d+\.) ... 「-」か「*」か「数字.」のどれかがある（箇条書きor番号付き）
// \s* ... スペース
// \[(.|x|X)\] ... [ ] か [x] か [X] がある
// gm ... 全行に対して実行
const regex = /^\s*(?:[-*+]|\d+\.)\s*\[(\s|x|X)\]/gm;

let total = 0;
let completed = 0;

// マッチした行をすべてチェック
const matches = content.match(regex);

if (matches) {
    total = matches.length;
    matches.forEach(m => {
        // [x] または [X] が含まれていたら完了とみなす
        if (m.includes("[x]") || m.includes("[X]")) {
            completed++;
        }
    });
}

// 残りの計算
const incomplete = total - completed;
const progress = total === 0 ? 0 : Math.round((completed / total) * 100);

// デバッグ表示（確認用）
dv.paragraph(`**完全スキャン結果:** ファイル全体から **${total}個** のチェックボックスを発見しました。（完了: ${completed} / 未完了: ${incomplete}）`);

// 3. グラフ描画
const chartData = {
    type: 'doughnut',
    data: {
        labels: ['完了 (' + completed + ')', '未完了 (' + incomplete + ')'],
        datasets: [{
            label: 'タスク状況',
            data: [completed, incomplete],
            backgroundColor: ['rgba(75, 192, 192, 0.6)', 'rgba(255, 99, 132, 0.6)'],
            borderColor: ['rgba(75, 192, 192, 1)', 'rgba(255, 99, 132, 1)'],
            borderWidth: 1
        }]
    },
    options: {
        responsive: true,
        maintainAspectRatio: false,
        plugins: {
            legend: { position: 'bottom' },
            title: {
                display: true,
                text: '進捗率: ' + progress + '% (' + completed + '/' + total + ')',
                font: { size: 16 }
            }
        }
    }
};

window.renderChart(chartData, this.container);
this.container.style.height = "300px";
```
