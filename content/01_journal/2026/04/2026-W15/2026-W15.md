// 1. 起点となる日付を確実に取得
const targetDate = moment("2026-04-06", "YYYY-MM-DD");

// 数学的に「今週の月曜日」を特定する (updateLocaleに依存しない)
// 日曜(0)なら6日前、月曜(1)なら0日前、土曜(6)なら5日前を引く
const dayNum = targetDate.day();
const diffToMon = dayNum === 0 ? 6 : dayNum - 1;
const startOfThisWeek = targetDate.clone().subtract(diffToMon, 'days').startOf('day');
const endOfThisWeek = startOfThisWeek.clone().add(6, 'days').endOf('day');

// デバッグ用（データが出ない時、ここをアンコメントすると期間が表示されます）
// dv.paragraph(`集計期間: ${startOfThisWeek.format("YYYY-MM-DD")} 〜 ${endOfThisWeek.format("YYYY-MM-DD")}`);

const daysOfWeek = ["MON", "TUE", "WED", "THU", "FRI", "SAT", "SUN"];

const pages = dv.pages("#dailyjournal")
  .filter(p => {
    const date = moment(p.file.name, "YYYY-MM-DD");
    // 厳密な比較。時刻を無視して日付のみで判定
    return date.isSameOrAfter(startOfThisWeek, 'day') && date.isSameOrBefore(endOfThisWeek, 'day');
  })
  .filter(p => p.file.name !== "temp_daily_journal")
  .sort(p => p.file.name);

const fields = [
  { key: "ExerciseTime", title: "EXERCISE" },
  { key: "StudyTime", title: "STUDY" },
  { key: "WorkTime", title: "WORK" },
  { key: "weight", title: "WEIGHT" },
  { key: "FatRate", title: "FAT RATE" },
];

function getValue(page, key) {
  if (key === "sleep") {
    if (!page.SleepFrom || !page.SleepTo) return null;
    const s = moment(page.SleepFrom.toString());
    const e = moment(page.SleepTo.toString());
    if (!s.isValid() || !e.isValid()) return null;
    const diff = moment.duration(e.diff(s)).asHours();
    return (diff > 0 && diff < 24) ? Math.round(diff * 10) / 10 : null;
  }
  return page[key];
}

function analyzeField(key) {
  const results = { sum: 0, count: 0, days: {} };
  daysOfWeek.forEach(d => results.days[d] = null);

  for (const p of pages) {
    const val = getValue(p, key);
    if (val != null && !isNaN(val)) {
      const date = moment(p.file.name, "YYYY-MM-DD");
      // テーブルのインデックス計算
      const dNum = date.day();
      const dowIndex = dNum === 0 ? 6 : dNum - 1; 
      const dow = daysOfWeek[dowIndex];
      
      results.sum += val;
      results.count += 1;
      results.days[dow] = val;
    }
  }

  return {
    SUM: results.sum ? (Math.round(results.sum * 10) / 10) : "",
    AVG: results.count ? (results.sum / results.count).toFixed(2) : "",
    ...results.days
  };
}

const table = [];
const sleepResult = analyzeField("sleep");
table.push(["SLEEP", sleepResult.SUM, sleepResult.AVG, ...daysOfWeek.map(d => sleepResult[d])]);

for (const field of fields) {
  const result = analyzeField(field.key);
  table.push([field.title, result.SUM, result.AVG, ...daysOfWeek.map(d => result[d])]);
}

dv.table(
  ["TITLE", "SUM", "AVG", "MON", "TUE", "WED", "THU", "FRI", "SAT", "SUN"],
  table
);

    const d = moment(p.file.name, "YYYY-MM-DD");
    return d.isSameOrAfter(startOfThisWeek, 'day') && d.isSameOrBefore(endOfThisWeek, 'day');

  .filter(p => p.file.name !== "temp_daily_journal")
  .sort(p => p.file.name);

// 2. 睡眠時間の計算関数
function calculateSleepDuration(from, to) {
  if (!from || !to) return 0;
  const start = moment(from.toString());
  const end = moment(to.toString());
  if (!start.isValid() || !end.isValid()) return 0;
  const hours = moment.duration(end.diff(start)).asHours();
  return (hours > 0 && hours < 24) ? Math.round(hours * 10) / 10 : 0;

// 3. 7日間分の配列を生成（データがない日を埋める）

const moodData = [];
const weightData = [];

// 1. 期間の設定 (今週と先週)
const targetDate = moment("2026-04-06", "YYYY-MM-DD");
const dayNum = targetDate.day();
const diffToMon = dayNum === 0 ? 6 : dayNum - 1;

// 今週の月曜〜日曜
const startThisWeek = targetDate.clone().subtract(diffToMon, 'days').startOf('day');
const endThisWeek = startThisWeek.clone().add(6, 'days').endOf('day');

// 先週の月曜〜日曜
const startLastWeek = startThisWeek.clone().subtract(7, 'days');
const endLastWeek = startLastWeek.clone().add(6, 'days').endOf('day');

// 2. データ取得関数（NaN対策・防衛型）
function getWeeklyFreeTime(start, end) {
    const pages = dv.pages('#dailyjournal').filter(p => {
        const d = moment(p.file.name, "YYYY-MM-DD");
        return d.isSameOrAfter(start, 'day') && d.isSameOrBefore(end, 'day');
    });

    let totalFree = 0;
    let daysRecorded = 0;

    pages.forEach(p => {
        // 全ての値を強制的に数値に変換し、変換できない場合は0にする
        const toNum = (val) => {
            const n = parseFloat(val);
            return isNaN(n) ? 0 : n;
        };

        const sleep = (() => {
            if (!p.SleepFrom || !p.SleepTo) return 0;
            const s = moment(p.SleepFrom.toString());
            const e = moment(p.SleepTo.toString());
            if (!s.isValid() || !e.isValid()) return 0;
            const diff = moment.duration(e.diff(s)).asHours();
            return isNaN(diff) ? 0 : Math.max(0, diff);
        })();

        const work = toNum(p.WorkTime);
        const study = toNum(p.StudyTime);
        const exercise = toNum(p.ExerciseTime);
        
        // 1日分の自由時間を計算
        const dailyFree = Math.max(0, 24 - (sleep + work + study + exercise));
        
        totalFree += dailyFree;
        daysRecorded++;
    });

    return { 
        total: isNaN(totalFree) ? 0 : totalFree, 
        avg: daysRecorded ? (totalFree / daysRecorded) : 0, 
        count: daysRecorded 
    };
}

// 3. 集計実行
const thisWeek = getWeeklyFreeTime(startThisWeek, endThisWeek);
const lastWeek = getWeeklyFreeTime(startLastWeek, endLastWeek);

// 4. 分析コメントの生成
const diff = thisWeek.total - lastWeek.total;
const diffStr = diff >= 0 ? `+${diff.toFixed(1)}` : `${diff.toFixed(1)}`;
const color = diff >= 0 ? "lightgreen" : "coral";

let comment = "";
if (thisWeek.count === 0) {
    comment = "今週のデータがまだ入力されていないようです。";
} else if (lastWeek.count === 0) {
    comment = "先週のデータがないため比較できませんが、今週はしっかり記録できていますね！";
} else {
    if (diff > 5) {
        comment = `先週より **${diffStr}時間** も自由時間が増えました！余裕が生まれています。`;
    } else if (diff < -5) {
        comment = `先週より **${diffStr}時間** 自由時間が減っています。少し忙しすぎるかもしれません。`;
    } else {
        comment = "先週とほぼ変わらないペースです。安定したリズムで過ごせています。";
    }
}

// 5. 表示 (HTML/CSS)
const summaryContainer = this.container.createEl("div", {
    style: "background-color: rgba(100, 100, 100, 0.1); padding: 15px; border-radius: 10px; margin-top: 20px; border-left: 5px solid " + color
});

summaryContainer.createEl("h3", { text: "⏳ グレー領域（自由時間）の分析", style: "margin-top: 0;" });
summaryContainer.createEl("p", { 
    text: `今週の合計自由時間: ${thisWeek.total.toFixed(1)}h （先週比: ${diffStr}h）`,
    style: "font-size: 1.1em; font-weight: bold;"
});
summaryContainer.createEl("p", { text: comment, style: "font-style: italic; color: var(--text-muted);" });

// --- (ここに前回のグラフ描画コードを続けると、分析の後にグラフが表示されます) ---
