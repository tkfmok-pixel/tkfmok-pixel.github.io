const today = moment("2026-01-01", "YYYY-MM-DD").startOf("day");
const sevenDaysAgo = today.clone().subtract(6, "days");

const pages = dv.pages('#dailyjournal')
  .filter(p => {
    const date = moment(p.file.name, "YYYY-MM-DD");
    return date.isBetween(sevenDaysAgo, today.clone().add(1, "day"), null, '[)');
  })
  .sort(p => p.file.name);

function calculateSleepDuration(from, to) {
  if (!from || !to) return 0;
  const start = moment(from.toString());
  const end = moment(to.toString());
  if (!start.isValid() || !end.isValid()) return 0;
  const hours = moment.duration(end.diff(start)).asHours();
  return (hours > 0 && hours < 24) ? Math.round(hours * 10) / 10 : 0;
}

function extract(pages, key, defaultValue = 0) {
  return pages.map(p => (p[key] !== undefined && p[key] !== null && p[key] !== "") ? p[key] : defaultValue).values;
}

const days = pages.map(p => p.file.name).values;
const mood = extract(pages, 'mood', 5); // 10段階なのでデフォルトを5に
const weight = extract(pages, 'weight', null);
const sleep = pages.map(p => calculateSleepDuration(p.SleepFrom, p.SleepTo)).values;

const chartData = {
  type: 'line',
  data: {
    labels: days,
    datasets: [
      {
        label: 'Sleep (h)',
        data: sleep,

        tension: 0.3
      },

        label: 'Mood',

      }
    ]
  },
  options: {
    responsive: true,
    scales: {
      y: { // 左: 睡眠
        type: 'linear',
        position: 'left',

  }
};

const fileDateStr = dv.current().file.name;
const targetDate = moment(fileDateStr, "YYYY-MM-DD").startOf("day");
const nextDate = moment(targetDate).add(1, "day");

dv.list(
  dv.pages('"Clippings"')
    .where(p => p.file.ctime >= targetDate && p.file.ctime < nextDate)
    .sort(p => p.file.ctime, 'asc')
    .map(p => {
      const linkTitle = `[[${p.file.name}]]：`;
      return `${linkTitle}${p.description ?? ""}<br><br><img src="${p.image ?? ""}" width="400px" height="200px">`;

# 🚣TIMELINE🚣
```dataviewjs
const fileDateStr = dv.current().file.name;
const targetDate = moment(fileDateStr, "YYYY-MM-DD").startOf("day");
const nextDate = moment(targetDate).add(1, "day");

const imageExtensions = [".png", ".jpg", ".jpeg", ".webp", ".gif", ".bmp"];
dv.table(["ファイル", "作成日時"],
  dv.pages()
    .where(p =>
      p.file.ctime >= targetDate &&
      p.file.ctime < nextDate &&
      !p.file.path.includes("Clippings") &&
      !imageExtensions.some(ext => p.file.name.toLowerCase().endsWith(ext))
    )
    .sort(p => p.file.ctime, 'asc')
    .map(p => [
      p.file.link,
      `${p.file.ctime.toFormat("yyyy年M月d日 H時mm分")}`
    ])
);
```
