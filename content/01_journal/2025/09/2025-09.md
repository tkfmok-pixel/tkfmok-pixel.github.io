> [!IMPORTANT]+ 📊Important Index📊
>```dataviewjs
>// === 対象月（Templaterで埋め込む）===
>const targetMonth = "2025-09"; // 例: "2025-06"
>
>// === 対象月の開始・終了日を生成 ===
>const startOfMonth = moment(targetMonth + "-01", "YYYY-MM-DD");
>const endOfMonth = startOfMonth.clone().endOf("month");
>
>// === 対象ノート取得（dailyjournalタグ、日付が月内）===
>const pages = dv.pages("#dailyjournal")
>  .where(p => {
>    const date = moment(p.file.name, "YYYY-MM-DD");
>    return date.isBetween(startOfMonth, endOfMonth, null, '[]');
> })
>  .where(p => p.file.name !== "temp_daily_journal")
>  .sort(p => p.file.name);
>
>// === 対象フィールド ===
>const fields = [
>  { key: "sleep", title: "SLEEP" },
>  { key: "exercise", title: "EXERCISE" },
>  { key: "reading/study", title: "READING/STUDY" },
>  { key: "weight", title: "WEIGHT" },
>  { key: "FatRate", title: "FAT RATE" },
>];
>
>// === 週別データのマッピング ===
>const weekMap = {};  // { "2025-W25": { sleep: [6.5, ...], ... } }
>for (const p of pages) {
>  const date = moment(p.file.name, "YYYY-MM-DD");
>  const weekLabel = date.format("GGGG-[W]WW");
>
>  if (!weekMap[weekLabel]) weekMap[weekLabel] = {};
>
>  for (const field of fields) {
>    const val = p[field.key];
>    if (val != null && !isNaN(val)) {
>      if (!weekMap[weekLabel][field.key]) weekMap[weekLabel][field.key] = [];
>      weekMap[weekLabel][field.key].push(val);
>    }
>  }
>}
>
>// === ソート済みの週リスト ===
>const sortedWeeks = Object.keys(weekMap).sort();
>
>// === 表データ構築 ===
>const table = fields.map(field => {
>  let sum = 0;
>  let count = 0;
>  const row = [field.title];
>
>  for (const week of sortedWeeks) {
>    const vals = weekMap[week][field.key] || [];
>    sum += vals.reduce((a, b) => a + b, 0);
>    count += vals.length;
> }
>
>  row.push(sum || "");
>  row.push(count ? (sum / count).toFixed(2) : "");
>
>  for (const week of sortedWeeks) {
>    const vals = weekMap[week][field.key] || [];
>    const avg = vals.length ? (vals.reduce((a, b) => a + b, 0) / vals.length).toFixed(2) : "";
>    row.push(avg);
>  }
>
>  return row;
>});
>
>// === ヘッダー出力 ===
>const header = ["TITLE", "SUM", "AVG", ...sortedWeeks];
>
>dv.table(header, table);
>```

>```dataviewjs
>// === 今月のYYYY-MMを取得 ===
const currentMonth =  "2025-09";
>
>// === タスクの抽出と構造化 ===
>const allTasksNested = dv.pages("").file.tasks
 > .filter(task => task.text && task.text.includes("✅") && task.text.includes(";"))
 > .map(task => {
>    const dateMatch = task.text.match(/✅ (\d{4}-\d{2}-\d{2})/);
>    const minutesMatch = task.text.match(/;(\d+)/);
>    const tagsMatch = task.text.match(/#([a-zA-Z0-9\u3000-\u30FF\u4E00-\u9FFF_\/]+)/g);
>
 >   if (!dateMatch || !minutesMatch || !tagsMatch) return [];
>
>    const dateStr = dateMatch[1]; // "YYYY-MM-DD"
 >   if (!dateStr.startsWith(currentMonth)) return []; // 今月以外は除外
>
 >  const date = new Date(dateStr);
>    const year = date.getFullYear();
>    const start = new Date(year, 0, 1);
>    const days = (date - start) / (1000 * 60 * 60 * 24);
>    const week = Math.ceil((days + start.getDay()) / 7);
>    const weekLabel = `${year}-W${String(week).padStart(2, "0")}`;
>
 >   const minutes = parseInt(minutesMatch[1]);
>
>    // タグからmain/subの組を抽出
>    const tagPairs = tagsMatch.map(tag => {
>      const parts = tag.replace("#", "").split("/");
>      return {
>        main: parts[0] || "N/A",
 >       sub: parts.length > 1 ? parts.slice(1).join("/") : "N/A"
>      };
 >   });
>
 >   return tagPairs.map(pair => ({
>      week: weekLabel,
 >     main: pair.main,
>      sub: pair.sub,
 >     minutes
>    }));
>  });
>
>// flat() の代替
>const allTasks = [].concat(...allTasksNested);
>
>// === 集計：週 × 分類 × 区分 ===
>const grouped = {};
>for (const task of allTasks) {
>  const key = `${task.week}||${task.main}||${task.sub}`;
>  if (!grouped[key]) {
>    grouped[key] = {
>      week: task.week,
>      main: "[[" + task.main + "]]",
>      sub: "[[" + task.sub + "]]",
>      total: 0
>    };
>  }
>  grouped[key].total += task.minutes;
>}
>
>// === 表形式に変換 & 表示 ===
>const rows = Object.values(grouped)
>  .sort((a, b) =>
>    a.week.localeCompare(b.week) ||
>    a.main.localeCompare(b.main) ||
>    a.sub.localeCompare(b.sub)
 > )
 > .map(entry => [entry.week, entry.main, entry.sub, entry.total]);
>
>dv.table(["週", "分類（メインタグ）", "区分（サブタグ）", "合計時間（分）"], rows);
>```
