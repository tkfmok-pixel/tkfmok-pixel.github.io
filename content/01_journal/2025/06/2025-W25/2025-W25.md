>// 日付が「2024年39週」に該当するかを確認する関数
>function isInWeek(taskDate, targetYear, targetWeek) {
 >   const firstDayOfYear = new Date(targetYear, 0, 1);
 >   const pastDaysOfYear = (taskDate - firstDayOfYear) / 86400000; // タスクの完了日までの経過日数
>    const taskWeek = Math.ceil((pastDaysOfYear + firstDayOfYear.getDay() ) / 7); // タスクが完了した週番号を計算
>
 >   return taskDate.getFullYear() === targetYear && taskWeek === targetWeek;
>}
>
>const tasks = dv.pages("").file.tasks
 >   .filter(task => task.text.includes(";")) // ";"を含むタスクをフィルタリング
 >   .where(task => {

 >       if (match) {
 >           const taskCompletedDate = new Date(match[1]); // 完了日を日付オブジェクトに変換
>            return isInWeek(taskCompletedDate, targetYear, targetWeek); // 該当週に完了したかを確認
>        }
 >       return false; // 完了日がない場合は無視
 >   });
>
>for (let i = 0; i < tasks.length; i++) {
 >   const task = tasks[i];
 >   if (task.text.includes(";")) {
 >       const taskText = task.text;

>        if (parts) {
 >           const [_, startTime, taskName, tag, actualTime] = parts;
 >           inputString += Number(actualTime); // 実績時間を集計
 >       }
 >   }
>}
>
>const roundedInputString = Math.round((inputString / 60) * 10) / 10;
>
>dv.span("Recorded <span style='color:red; font-weight:bold;'>" + roundedInputString + "</span> hours of worktime in <span style='color:blue; font-weight:bold;'> THIS WEEK </span>");

> # IN DETAIL
>```dataviewjs
>let inputString = 0;
>
>const targetYear =  2025; // 対象年
>const targetWeek = 25;   // 対象週
>
>// 日付が「2024年39週」に該当するかを確認する関数
>function isInWeek(taskDate, targetYear, targetWeek) {
>    const firstDayOfYear = new Date(targetYear, 0, 1);
>    const pastDaysOfYear = (taskDate - firstDayOfYear) / 86400000; // タスクの完了日までの経過日数
>    const taskWeek = Math.ceil((pastDaysOfYear + firstDayOfYear.getDay() ) / 7); // タスクが完了した週番号を計算
>
  >  return taskDate.getFullYear() === targetYear && taskWeek === targetWeek;
>}
>
>const tasks = dv.pages("").file.tasks
 >   .filter(task => task.text && task.text.includes(";")) // ";"を含むタスクかつ、textが存在するか確認
>    .where(task => {
 >       if (!task.text) return false; // タスクのテキストが存在しない場合は無視
>
 >       const match = task.text.match(/✅ (\d{4}-\d{2}-\d{2})/); // 完了日を抽出
 >       if (match) {
 >           const taskCompletedDate = new Date(match[1]); // 完了日を日付オブジェクトに変換
>            return isInWeek(taskCompletedDate, targetYear, targetWeek); // 該当週に完了したかを確認
 >       }
 >       return false; // 完了日がない場合は無視
 >   })
 >   .sort((a, b) => {
 >       if (!a.text || !b.text) return 0; // テキストがない場合は同順に扱う
>
 >       const dateA = a.text.match(/✅ (\d{4}-\d{2}-\d{2})/) ? new Date(a.text.match(/✅ (\d{4}-\d{2}-\d{2})/)[1]) : new Date();
>    const dateB = b.text.match(/✅ (\d{4}-\d{2}-\d{2})/) ? new Date(b.text.match(/✅ (\d{4}-\d{2}-\d{2})/)[1]) : new Date();
>
  >      return dateA - dateB; // 昇順でソート（小さい日付が先にくる）
 >   });
>
>// タスクの情報を抽出し、テーブルとして表示
> dv.table(["完了日", "タスク", "所要（分）", "分類", "区分"],
>    tasks
>        .filter(task => task.text && task.text.includes("✅")) // 完了マークがあるタスクに絞り込み
>	.sort((a, b) => {
>    const dateMatchA = a.text ? a.text.match(/✅ (\d{4}-\d{2}-\d{2})/) : null;
>    const dateMatchB = b.text ? b.text.match(/✅ (\d{4}-\d{2}-\d{2})/) : null;
>    const dateA = dateMatchA ? new Date(dateMatchA[1]) : new Date(0); // 日付がない場合は古い日付を設定
>    const dateB = dateMatchB ? new Date(dateMatchB[1]) : new Date(0);
>    return dateA - dateB;
> })
>        .map(task => {
>            if (!task.text) return ["N/A", "N/A", "N/A", "N/A", "N/A"];
>
>            // 完了日を抽出
>            const match = task.text.match(/✅ (\d{4}-\d{2}-\d{2})/);
>            const completionDate = match ? match[1] : "N/A";
>            const filelink = "[[" + completionDate + "]]";
>
>            // タスク名から時間部分と完了日を削除し、タグも削除
>            const cleanedTaskText = task.text
>                .replace(/\d{1,2}:\d{2}\s?-\s?\d{1,2}:\d{2}/g, "")
>                .replace(/✅ (\d{4}-\d{2}-\d{2})/g, "")
>                .replace(/;(\d+)/g, "")
>                .replace(/#[a-zA-Z0-9\u3000-\u30FF\u4E00-\u9FFF_\/]+/g, "")
>                .trim();
>
>            // タスク名から所要時間を抽出し、「所要（分）」カラムとして表示
>            const minutesMatch = task.text.match(/;(\d+)/);
>            const minutes = minutesMatch ? minutesMatch[1] : "N/A";
>
>            // タスク名からタグ（階層タグを含む）を抽出
>            const tagsMatch = task.text.match(/#([a-zA-Z0-9\u3000-\u30FF\u4E00-\u9FFF_\/]+)/g);
>
>            // 分類（メインタグ）を抽出
>            const mainTag = tagsMatch && tagsMatch.length > 0 ? tagsMatch.map(tag => {
>                const parts = tag.split('/');
>                return parts[0].replace("#", "");
>            }).join(", ") : "N/A";
>            const category = "[[" + mainTag + "]]";
>
>            // サブタグ（区分）を抽出
>            const subTags = tagsMatch && tagsMatch.length > 0 ? tagsMatch.map(tag => {
>                const parts = tag.split('/');
>                return parts.length > 1 ? parts.slice(1).join("/") : "";
>            }).filter(subTag => subTag).join(", ") : "N/A";
>            const type = "[[" + subTags + "]]";
>
>            return [filelink, cleanedTaskText, minutes, category, type];
>        })
> );
>```
