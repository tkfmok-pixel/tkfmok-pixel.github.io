const start = moment("2025-11-15", "YYYY-MM-DD").startOf("day");
const end = start.clone().subtract(6, "days");

const pages = dv.pages("#dailyjournal")
  .where(p => {
    const date = moment(p.file.name, "YYYY-MM-DD");
    return date.isBetween(end.clone().subtract(1, "day"), start.clone().add(1, "day"));
  })
  .where(p => p.file.name !== "temp_daily_journal")
  .sort(p => p.file.name);

const daysOfWeek = ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"];
const fields = [
  { key: "sleep", title: "SLEEP" },
  { key: "exercise", title: "EXERCISE" },
  { key: "reading/study", title: "READING/STUDY" },
  { key: "weight", title: "WEIGHT" },
  { key: "FatRate", title: "FAT RATE" },
];

function analyzeField(key) {
  const results = { sum: 0, count: 0, days: {} };
  daysOfWeek.forEach(d => results.days[d] = null);

  for (const p of pages) {
    const val = p[key];
    if (val != null && !isNaN(val)) {
      const date = moment(p.file.name, "YYYY-MM-DD");
      const dow = daysOfWeek[date.day()];
      results.sum += val;
      results.count += 1;
      results.days[dow] = val;
    }
  }

  return {
    SUM: results.sum || "",
    AVG: results.count ? (results.sum / results.count).toFixed(2) : "",
    ...results.days
  };
}

const table = [];

for (const field of fields) {
  const result = analyzeField(field.key);
  table.push([
    field.title,
    result.SUM,
    result.AVG,
    result["SUN"],
    result["MON"],
    result["TUE"],
    result["WED"],
    result["THU"],
    result["FRI"],
    result["SAT"],
  ]);
}

// ヘッダーとテーブルを出力
dv.table(
  ["TITLE", "SUM", "AVG", "SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
  table
);

let totalMinutes = 0;
const targetYear = 2025; // 対象年
const targetWeek = 46;    // 対象週
// ==== 週番号で絞り込み ====
function isInWeek(taskDate, targetYear, targetWeek) {
  const firstDayOfYear = new Date(targetYear, 0, 1);
  const pastDaysOfYear = (taskDate - firstDayOfYear) / 86400000;
  const taskWeek = Math.ceil((pastDaysOfYear + firstDayOfYear.getDay()) / 7);
  return taskDate.getFullYear() === targetYear && taskWeek === targetWeek;
}

// ==== タスク抽出 ====
const tasks = dv.pages("").file.tasks
  .filter(task => task.text.includes(";"))
  .where(task => {
    const match = task.text.match(/✅ (\d{4}-\d{2}-\d{2})/);
    if (match) {
      const taskCompletedDate = new Date(match[1]);
      return isInWeek(taskCompletedDate, targetYear, targetWeek);
    }
    return false;
  });

// ==== 実績合計時間（分→時間） ====
const tagMinuteMap = {};
for (let task of tasks) {
  const parts = task.text.match(/(\d{2}:\d{2})\s*(.*?)\s*#(.*?)\s*;\s*(\d+)/);
  if (parts) {
    const [_, startTime, taskName, tag, actualMinutes] = parts;
    const minutes = Number(actualMinutes);
    totalMinutes += minutes;
    const tagName = tag.trim();
    if (!tagMinuteMap[tagName]) tagMinuteMap[tagName] = 0;
    tagMinuteMap[tagName] += minutes;
  }
}
const totalHours = Math.round((totalMinutes / 60) * 10) / 10;

// ==== 結果表示 ====
dv.span(`Recorded <span style='color:red; font-weight:bold;'>${totalHours}</span> hours of worktime in <span style='color:blue; font-weight:bold;'>THIS WEEK</span>`);

// ==== グラフ用データ準備 ====
const tagTimePairs = Object.entries(tagMinuteMap)
  .map(([tag, minutes]) => [tag, Math.round((minutes / 60) * 10) / 10])
  .sort((a, b) => b[1] - a[1]);

const sortedTags = tagTimePairs.map(([tag, _]) => tag);
const sortedValues = tagTimePairs.map(([_, time]) => time);

// ==== 色指定 ====

let inputString = 0;

const targetYear =  2025; // 対象年
const targetWeek = 46;   // 対象週

function isInWeek(taskDate, targetYear, targetWeek) {
    const firstDayOfYear = new Date(targetYear, 0, 1);
    const pastDaysOfYear = (taskDate - firstDayOfYear) / 86400000;
    const taskWeek = Math.ceil((pastDaysOfYear + firstDayOfYear.getDay() ) / 7);

    return taskDate.getFullYear() === targetYear && taskWeek === targetWeek;
}

const tasks = dv.pages("").file.tasks
    .filter(task => task.text && task.text.includes(";"))
    .where(task => {
        if (!task.text) return false;

        const match = task.text.match(/✅ (\d{4}-\d{2}-\d{2})/);
        if (match) {
            const taskCompletedDate = new Date(match[1]);
            return isInWeek(taskCompletedDate, targetYear, targetWeek);
        }
        return false;
    })
    .sort((a, b) => {
        if (!a.text || !b.text) return 0;

        const dateA = a.text.match(/✅ (\d{4}-\d{2}-\d{2})/) ? new Date(a.text.match(/✅ (\d{4}-\d{2}-\d{2})/)[1]) : new Date();
        const dateB = b.text.match(/✅ (\d{4}-\d{2}-\d{2})/) ? new Date(b.text.match(/✅ (\d{4}-\d{2}-\d{2})/)[1]) : new Date();

        return dateA - dateB;
    });

dv.table(["完了日", "タスク", "所要（分）", "分類", "区分"],
    tasks
        .filter(task => task.text && task.text.includes("✅"))
        .sort((a, b) => {
            const dateMatchA = a.text ? a.text.match(/✅ (\d{4}-\d{2}-\d{2})/) : null;
            const dateMatchB = b.text ? b.text.match(/✅ (\d{4}-\d{2}-\d{2})/) : null;
            const dateA = dateMatchA ? new Date(dateMatchA[1]) : new Date(0);
            const dateB = dateMatchB ? new Date(dateMatchB[1]) : new Date(0);
            return dateA - dateB;
        })
        .map(task => {
            if (!task.text) return ["N/A", "N/A", "N/A", "N/A", "N/A"];

            const match = task.text.match(/✅ (\d{4}-\d{2}-\d{2})/);
            const completionDate = match ? match[1] : "N/A";
            const filelink = "[[" + completionDate + "]]";

            const cleanedTaskText = task.text
                .replace(/\d{1,2}:\d{2}\s?-\s?\d{1,2}:\d{2}/g, "")
                .replace(/✅ (\d{4}-\d{2}-\d{2})/g, "")
                .replace(/;(\d+)/g, "")
                .replace(/#[a-zA-Z0-9\u3000-\u30FF\u4E00-\u9FFF_\/]+/g, "")
                .trim();

            const minutesMatch = task.text.match(/;(\d+)/);
            const minutes = minutesMatch ? minutesMatch[1] : "N/A";

            const tagsMatch = task.text.match(/#([a-zA-Z0-9\u3000-\u30FF\u4E00-\u9FFF_\/]+)/g);

            const mainTag = tagsMatch && tagsMatch.length > 0 ? tagsMatch.map(tag => {
                const parts = tag.split('/');
                return parts[0].replace("#", "");
            }).join(", ") : "N/A";
            const category = "[[" + mainTag + "]]";

            const subTags = tagsMatch && tagsMatch.length > 0 ? tagsMatch.map(tag => {
                const parts = tag.split('/');
                return parts.length > 1 ? parts.slice(1).join("/") : "";
            }).filter(subTag => subTag).join(", ") : "N/A";
            const type = "[[" + subTags + "]]";

            return [filelink, cleanedTaskText, minutes, category, type];
        })
);
